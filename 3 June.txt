package mongo;

import com.mongodb.client.*;
import com.mongodb.client.model.Filters;
import com.mongodb.client.model.Updates;
import org.bson.Document;

public class MongoDBConnection {

    public static void main(String[] args) {

        // Connect to MongoDB
        MongoClient mongoClient = MongoClients.create("mongodb://localhost:27017");

        // Access database
        MongoDatabase database = mongoClient.getDatabase("vitcampus");

        // Access collection
        MongoCollection<Document> collection =
                database.getCollection("StudentDetails");

        // ==========================
        // CREATE (Insert)
        // ==========================
        Document student = new Document("id", 37)
                .append("name", "Rohan Sharma")
                .append("department", "CSE")
                .append("cgpa", 9.1);

        collection.insertOne(student);
        System.out.println("Document Inserted Successfully");

        // ==========================
        // READ (Find)
        // ==========================
        System.out.println("\nStudent Records:");

        FindIterable<Document> students = collection.find();

        for (Document doc : students) {
            System.out.println(doc.toJson());
        }

        // ==========================
        // UPDATE
        // ==========================
        collection.updateOne(
                Filters.eq("id", 37),
                Updates.set("cgpa", 9.5)
        );

        System.out.println("\nDocument Updated Successfully");

        // ==========================
        // READ Updated Record
        // ==========================
        Document updatedStudent =
                collection.find(Filters.eq("id", 37)).first();

        System.out.println("\nUpdated Record:");
        System.out.println(updatedStudent.toJson());

        // ==========================
        // DELETE
        // ==========================
        collection.deleteOne(Filters.eq("id", 37));

        System.out.println("\nDocument Deleted Successfully");

        // Close connection
        mongoClient.close();
    }
}

package mongo;

import com.mongodb.MongoClient;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;
import com.mongodb.client.FindIterable;
import com.mongodb.client.model.Filters;
import com.mongodb.client.model.Updates;
import org.bson.Document;

public class MongoDBConnection {

    public static void main(String[] args) {

        MongoClient mongoClient = new MongoClient("localhost", 27017);

        MongoDatabase database = mongoClient.getDatabase("vitcampus");

        MongoCollection<Document> collection =
                database.getCollection("StudentDetails");

        // INSERT MULTIPLE RECORDS
        collection.insertMany(java.util.Arrays.asList(
                new Document("rollno", 201)
                        .append("name", "Rohan Sharma")
                        .append("branch", "CSE")
                        .append("cgpa", 9.1),

                new Document("rollno", 202)
                        .append("name", "Gaurav Kakkar")
                        .append("branch", "CSE")
                        .append("cgpa", 8.7),

                new Document("rollno", 203)
                        .append("name", "Sambhav Trivedi")
                        .append("branch", "ECE")
                        .append("cgpa", 7.4)
        ));

        System.out.println("Records Inserted Successfully");

        // DISPLAY ALL RECORDS
        System.out.println("\nAll Students:");
        FindIterable<Document> students = collection.find();

        for (Document doc : students) {
            System.out.println(doc.toJson());
        }

        // FIND PARTICULAR STUDENT
        Document student =
                collection.find(Filters.eq("rollno", 202)).first();

        System.out.println("\nStudent Found:");
        System.out.println(student.toJson());

        // UPDATE NAME
        collection.updateOne(
                Filters.eq("rollno", 202),
                Updates.set("name", "Abhay"));

        // UPDATE CGPA
        collection.updateOne(
                Filters.eq("rollno", 203),
                Updates.set("cgpa", 8.1));

        System.out.println("\nRecords Updated");

        // REPLACE COMPLETE DOCUMENT
        Document newStudent = new Document("rollno", 203)
                .append("name", "Arpit")
                .append("branch", "AIML")
                .append("cgpa", 9.3);

        collection.replaceOne(
                Filters.eq("rollno", 203),
                newStudent);

        System.out.println("\nDocument Replaced");

        // DELETE ONE RECORD
        collection.deleteOne(
                Filters.eq("rollno", 201));

        System.out.println("\nRecord Deleted");

        // FINAL RECORDS
        System.out.println("\nRemaining Records:");

        FindIterable<Document> finalStudents =
                collection.find();

        for (Document doc : finalStudents) {
            System.out.println(doc.toJson());
        }

        mongoClient.close();
    }
}
